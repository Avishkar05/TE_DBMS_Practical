-- Create employee table
CREATE TABLE emp (
  emp_id INT PRIMARY KEY,
  emp_name VARCHAR(50),
  job VARCHAR(30),
  manager_id INT,
  hire_date DATE,
  salary DECIMAL(10, 2),
  dept_id INT
);

-- Create department table
CREATE TABLE dept (
  dept_id INT PRIMARY KEY,
  dept_name VARCHAR(50),
  location VARCHAR(50)
);

-- Create a view for IT department employees
CREATE VIEW it_employees AS SELECT emp_name, job FROM emp WHERE dept_id = 10;

-- Query the view
