create database Joins;
use Joins;

-- Create first table
CREATE TABLE tableA (
  A_id INT PRIMARY KEY,
  A_name VARCHAR(50),
  A_salary DECIMAL(10,2)
);

-- Insert data into tableA
INSERT INTO tableA VALUES (1, 'Alice', 50000);
INSERT INTO tableA VALUES (2, 'Bob', 55000);
INSERT INTO tableA VALUES (3, 'Charlie', 60000);
INSERT INTO tableA VALUES (4, 'David', 40000);
INSERT INTO tableA VALUES (5, 'Eva', 70000);

-- Create second table
CREATE TABLE tableB (
  B_id INT PRIMARY KEY,
  B_name VARCHAR(50),
  A_id INT  -- this refers to tableA.A_id
);

-- Insert data into tableB
INSERT INTO tableB VALUES (101, 'IT', 1);
INSERT INTO tableB VALUES (102, 'HR', 2);
INSERT INTO tableB VALUES (103, 'Sales', 3);
INSERT INTO tableB VALUES (104, 'Marketing', 6); -- no match in tableA
INSERT INTO tableB VALUES (105, 'Finance', 5);

-- View both tables
SELECT * FROM tableA;
SELECT * FROM tableB;

-- INNER JOIN: only matching rows
SELECT a.A_id, a.A_name, b.B_name FROM tableA a INNER JOIN tableB b ON a.A_id = b.A_id;

-- LEFT JOIN: all rows from tableA + matches from tableB
SELECT a.A_id, a.A_name, b.B_name FROM tableA a LEFT JOIN tableB b ON a.A_id = b.A_id;

-- RIGHT JOIN: all rows from tableB + matches from tableA
SELECT a.A_id, a.A_name, b.B_name FROM tableA a RIGHT JOIN tableB b ON a.A_id = b.A_id;

-- FULL OUTER JOIN (MySQL doesn't support directly, use UNION)
SELECT a.A_id, a.A_name, b.B_name  FROM tableA a LEFT JOIN tableB b ON a.A_id = b.A_id
UNION
SELECT a.A_id, a.A_name, b.B_name FROM tableA a RIGHT JOIN tableB b ON a.A_id = b.A_id;

-- Find the person with maximum salary
SELECT A_name FROM tableA WHERE A_salary = (SELECT MAX(A_salary) FROM tableA);

-- Find names that exist in both tables
SELECT A_name FROM tableA WHERE A_id IN (SELECT A_id FROM tableB);

-- Create a view for employees with salary > 50000
CREATE OR REPLACE VIEW high_salary AS SELECT A_name, A_salary FROM tableA WHERE A_salary > 50000;

-- Show the view
SELECT * FROM high_salary;

-- Average salary
SELECT AVG(A_salary) AS avg_salary FROM tableA;

-- Count number of departments (tableB rows)
SELECT COUNT(B_id) AS total_B FROM tableB;

-- Sum of all salaries
SELECT SUM(A_salary) AS total_salary FROM tableA;
